JavaScript API

import { Combobox } from 'boodoo';

const element = document.getElementById('demoCombobox');
const combobox = Combobox.getOrCreateInstance(element);

// Listen to selection events
element.addEventListener('select.boodoo.combobox', (e) => {
  console.log('Selected value:', e.detail.value);
});

// Clear programmatically
combobox.clear();
