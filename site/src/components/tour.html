Tour

Step-by-step product walkthroughs, contextual feature callouts, backdrop highlights, and keyboard-driven onboarding tours.

Product Tour Demo

Tour Configuration

import { Tour } from 'boodoo';

const tour = new Tour({
  steps: [
    {
      target: '#nav-home',
      title: 'Navigation Header',
      content: 'Access all documentation and starters here.',
      placement: 'bottom'
    },
    {
      target: '#searchDocs',
      title: 'Global Search',
      content: 'Instantly find any utility, class, or component.',
      placement: 'bottom'
    }
  ]
});

document.getElementById('startTourBtn').addEventListener('click', () => {
  tour.start();
});
