2. index.php

<?php
/**
 * Main Template File - One Column Theme
 */
get_header(); ?>

<main class="container py-5">
  <div class="budu-one-col-container">
    <?php if ( have_posts() ) : while ( have_posts() ) : the_post(); ?>
      <article id="post-<?php the_ID(); ?>" <?php post_class( 'bg-white p-4 p-md-5 rounded-4 shadow-sm mb-5 border border-light' ); ?>>
        <header class="text-center mb-4">
          <div class="mb-2"><?php the_category( ' ' ); ?></div>
          <h1 class="display-5 fw-bold text-dark mb-3">
            <a href="<?php the_permalink(); ?>" class="text-dark text-decoration-none"><?php the_title(); ?></a>
          </h1>
          <div class="text-muted small">
            Posted on <?php echo get_the_date(); ?> by <?php the_author(); ?>
          </div>
        </header>
        <?php if ( has_post_thumbnail() ) : ?>
          <div class="mb-4 rounded-3 overflow-hidden">
            <?php the_post_thumbnail( 'large', array( 'class' => 'img-fluid w-100' ) ); ?>
          </div>
        <?php endif; ?>
        <div class="entry-content text-secondary fs-5" style="line-height: 1.8;">
          <?php the_content( 'Continue reading →' ); ?>
        </div>
      </article>
    <?php endwhile; endif; ?>
  </div>
</main>

<?php get_footer(); ?>
