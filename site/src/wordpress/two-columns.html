2. index.php

<?php
/**
 * Main Template File - Two Columns Theme
 */
get_header(); ?>

<main class="container py-5">
  <div class="row row-cols-1 row-cols-md-2 g-4">
    <?php if ( have_posts() ) : while ( have_posts() ) : the_post(); ?>
      <div class="col">
        <article id="post-<?php the_ID(); ?>" <?php post_class( 'card h-100 border-0 shadow-sm rounded-4 p-4 bg-white' ); ?>>
          <div class="mb-2"><?php the_category( ' ' ); ?></div>
          <h2 class="h4 card-title fw-bold mb-3">
            <a href="<?php the_permalink(); ?>" class="text-dark text-decoration-none"><?php the_title(); ?></a>
          </h2>
          <div class="card-text text-muted mb-4">
            <?php the_excerpt(); ?>
          </div>
          <div class="mt-auto">
            <a href="<?php the_permalink(); ?>" class="btn btn-outline-primary btn-sm rounded-pill">Read More →</a>
          </div>
        </article>
      </div>
    <?php endwhile; endif; ?>
  </div>
</main>

<?php get_footer(); ?>
