The Right Sidebar Theme provides the classic WordPress layout structure: primary content stream occupying 66% (8/12 columns) on the left paired with a 33% (4/12 columns) widget column on the right (col-lg-8 col-md-7 + col-lg-4 col-md-5).

2. index.php

<?php
/**
 * Main Template File - Right Sidebar Theme
 */
get_header(); ?>

<main class="container py-5">
  <div class="row g-4">
    <!-- Main Content Column -->
    <section class="col-lg-8 col-md-7">
      <?php if ( have_posts() ) : while ( have_posts() ) : the_post(); ?>
        <article id="post-<?php the_ID(); ?>" <?php post_class( 'card border-0 shadow-sm rounded-4 p-4 mb-4 bg-white' ); ?>>
          <h2 class="h3 card-title fw-bold mb-3">
            <a href="<?php the_permalink(); ?>" class="text-dark text-decoration-none"><?php the_title(); ?></a>
          </h2>
          <div class="card-text text-muted mb-4"><?php the_excerpt(); ?></div>
          <div><a href="<?php the_permalink(); ?>" class="btn btn-primary btn-sm rounded-pill">Read Article →</a></div>
        </article>
      <?php endwhile; endif; ?>
    </section>

    <!-- Right Sidebar Column -->
    <aside class="col-lg-4 col-md-5">
      <?php get_sidebar(); ?>
    </aside>
  </div>
</main>

<?php get_footer(); ?>
