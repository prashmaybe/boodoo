Theme Code Files

1. style.css

/*
Theme Name: Budu Grid Layout Theme
Theme URI: https://boodoo.dihadiwala.com/docs/wordpress/grid-layout.html
Author: Budu Framework Team
Author URI: https://boodoo.dihadiwala.com
Description: Responsive multi-card magazine and portfolio grid WordPress theme built with Budu CSS framework.
Version: 1.0.2
License: MIT
Text Domain: budu-grid
*/

@import url('https://boodoo.dihadiwala.com/dist/css/boodoo.min.css');

/* Custom overrides for Grid layout */
.budu-grid-card {
  transition: transform 0.2s ease-in-out;
}
.budu-grid-card:hover {
  transform: translateY(-4px);
}

2. index.php

<?php
/**
 * Main Template File - Grid Layout Theme
 */
get_header(); ?>

<main class="container py-5">
  <header class="text-center mb-5">
    <h1 class="display-4 fw-bold text-dark"><?php single_post_title(); ?></h1>
    <p class="lead text-muted"><?php bloginfo( 'description' ); ?></p>
  </header>

  <div class="row row-cols-1 row-cols-md-2 row-cols-lg-3 g-4">
    <?php if ( have_posts() ) : while ( have_posts() ) : the_post(); ?>
      <div class="col">
        <article id="post-<?php the_ID(); ?>" <?php post_class( 'card h-100 border-0 shadow-sm rounded-4 overflow-hidden budu-grid-card' ); ?>>
          <?php if ( has_post_thumbnail() ) : ?>
            <a href="<?php the_permalink(); ?>">
              <?php the_post_thumbnail( 'medium_large', array( 'class' => 'card-img-top img-fluid' ) ); ?>
            </a>
          <?php endif; ?>
          <div class="card-body p-4">
            <div class="mb-2"><?php the_category( ' ' ); ?></div>
            <h2 class="h5 card-title fw-bold mb-3">
              <a href="<?php the_permalink(); ?>" class="text-dark text-decoration-none"><?php the_title(); ?></a>
            </h2>
            <p class="card-text text-muted small"><?php echo wp_trim_words( get_the_excerpt(), 18 ); ?></p>
          </div>
          <div class="card-footer bg-white border-top-0 p-4 pt-0">
            <a href="<?php the_permalink(); ?>" class="btn btn-outline-primary btn-sm rounded-pill w-100">Read Article →</a>
          </div>
        </article>
      </div>
    <?php endwhile; else : ?>
      <div class="col-12 text-center py-5">
        <p class="lead text-muted">No posts found.</p>
      </div>
    <?php endif; ?>
  </div>
</main>

<?php get_footer(); ?>
