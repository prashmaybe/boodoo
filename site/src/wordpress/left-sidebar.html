2. index.php

<?php
/**
 * Main Template File - Left Sidebar Theme
 */
get_header(); ?>

<main class="container py-5">
  <div class="row g-4">
    <!-- Left Sidebar Column -->
    <aside class="col-lg-3 col-md-4">
      <?php get_sidebar( 'left' ); ?>
    </aside>

    <!-- Main Content Area -->
    <section class="col-lg-9 col-md-8">
      <?php if ( have_posts() ) : while ( have_posts() ) : the_post(); ?>
        <article id="post-<?php the_ID(); ?>" <?php post_class( 'card border-0 shadow-sm rounded-4 p-4 mb-4 bg-white' ); ?>>
          <h2 class="h3 card-title fw-bold mb-3">
            <a href="<?php the_permalink(); ?>" class="text-dark text-decoration-none"><?php the_title(); ?></a>
          </h2>
          <div class="card-text text-muted mb-4"><?php the_excerpt(); ?></div>
          <div><a href="<?php the_permalink(); ?>" class="btn btn-primary btn-sm rounded-pill">Read Article →</a></div>
        </article>
      <?php endwhile; endif; ?>
    </section>
  </div>
</main>

<?php get_footer(); ?>
