2. index.php

<?php
/**
 * Main Template File - Four Columns Theme
 */
get_header(); ?>

<main class="container py-5">
  <div class="row row-cols-1 row-cols-sm-2 row-cols-lg-4 g-4">
    <?php if ( have_posts() ) : while ( have_posts() ) : the_post(); ?>
      <div class="col">
        <article id="post-<?php the_ID(); ?>" <?php post_class( 'card h-100 border-0 shadow-sm rounded-4 p-3 bg-white' ); ?>>
          <h4 class="h6 card-title fw-bold mb-2">
            <a href="<?php the_permalink(); ?>" class="text-dark text-decoration-none"><?php the_title(); ?></a>
          </h4>
          <p class="card-text text-muted small mb-3"><?php echo wp_trim_words( get_the_excerpt(), 12 ); ?></p>
          <div class="mt-auto">
            <a href="<?php the_permalink(); ?>" class="btn btn-outline-primary btn-sm rounded-pill w-100">View →</a>
          </div>
        </article>
      </div>
    <?php endwhile; endif; ?>
  </div>
</main>

<?php get_footer(); ?>
